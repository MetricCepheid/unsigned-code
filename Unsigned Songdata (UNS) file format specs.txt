struct BarLine
{
	uint time;
	uint numBeats;	
}

uint barLinesCount;

BarLine[] barLines

uint trailingBeatLen;

byte HasBRE;//(bool) whether the song has a Big Rock Ending

uint BREStart, BREEnd;//in MS from start of song

uint numHarmonies;

struct Harmony
{
	uint start, end;
	int instruments;//bitwise instruments
}

Harmony[numHarmonies] harmonies;


//Instrument Chunk

char[3] InstrumentType;

uint numOverdrivePhrases

struct OverdrivePhrase
{
	uint time;
	uint len;
}

OverdrivePhrase[numOverdrivePhrases] spph;

if(instrument HasSolos)
{
	uint numSolos;
	
	struct Solo
	{
		uint time;
		uint len;
	}
	
	Solo[numSolos] solos;
}

if((instrument RPEnableType)&FILLS != 0)
{
	uint numFills;
	
	struct Fill
	{
		uint time;
		uint len;
	}
	
	Fill[numFills] fills;
}

struct Note
{
	long value;//bitwise, bits 1-NumTracks=frets, bit NumTracks=HOPOable
	if(instrument PitchShifts)
	{
		long endvalue;
	}
	uint time;//in MS from beginning of song
	if(instrument ContainsHeldNotes)
	{
		uint len;//0 for non-held notes
	}
	if(instrument ContainsText)
	{
		String text;
	}
}

struct Phrase
{
	if(instrument TypesOfPhrases != NONE)
	{
		uint time;//in MS from start
		byte type;
		bool rockpower;
	}
	uint numNotes;
	Note[numNotes] notes;
}

Struct DifficultySet
{
	byte diff;//3=Expert, 0=Easy, etc
	uint numPhrases;//if instrument TypesOfPhrases == NONE, it always = 1
	Phrase[numPhrases] phrases;
	uint[6] starScoreLevels;
}

DifficultySet[4] notedata;

{

}

Effect[numEffects] effects
