//Instrument Chunk

char[3] InstrumentType;

uint numRockPowerPhrases

struct RockPowerPhrase
{
	uint time;
	uint len;
}

RockPowerPhrase[numRockPowerPhrases] spph;

if(instrument HasSolos)
{
	uint numSolos;
	
	struct Solo
	{
		uint time;
		uint len;
	}
	
	Solo[numSolos] solos;
}

if((instrument RPEnableType)&FILLS != 0)
{
	uint numFills;
	
	struct Fill
	{
		uint time;
		uint len;
	}
	
	Fill[numFills] fills;
}

struct Note
{
	ulong value;//bitwise, bits 1-NumTracks=frets, bit NumTracks=HOPOable
	if(instrument PitchShifts)
	{
		ulong endvalue;
	}
	uint time;//in MS from beginning of song
	if(instrument ContainsHeldNotes)
	{
		uint len;//0 for non-held notes
	}
	if(instrument ContainsText)
	{
		String text;
	}
}

struct Phrase
{
	if(instrument TypesOfPhrases != NONE)
	{
		uint time;//in MS from start
		byte type;
		bool rockpower;
		if(instrument TypesOfPhrases & RHYTHM)
			byte rhythmType;
	}
	uint numNotes;
	Note[numNotes] notes;
}

Struct DifficultySet
{
	byte diff;//3=Expert, 0=Easy, etc
	uint numPhrases;//if instrument TypesOfPhrases == NONE, it always = 1
	Phrase[numPhrases] phrases;
	uint[6] starScoreLevels;
}

uint numDiffSets;

DifficultySet[numDiffSets] notedata;
