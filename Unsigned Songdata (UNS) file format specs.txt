struct Note
{
	long value;//bitwise, bits 1-NumTracks=frets, bit NumTracks=HOPOable
	if(instrument PitchShifts)
	{
		long endvalue;
	}
	int time;//in MS from beginning of song
	if(instrument ContainsHeldNotes)
	{
		int len;//0 for non-held notes
	}
	if(instrument ContainsText)
	{
		String text;
	}
}

struct Phrase
{
	if(instrument TypesOfPhrases != NONE)
	{
		uint time;//in MS from start
		byte type;
		bool rockpower;
	}
	int numNotes;
	Note[numNotes] notes;
}

Struct DifficultySet
{
	int diff;//3=Expert, 0=Easy, etc
	int numPhrases;//if instrument TypesOfPhrases == NONE, it always = 1
	Phrase[numPhrases] phrases;
	int[6] starScoreLevels;
}

DifficultySet[4] notedata;
