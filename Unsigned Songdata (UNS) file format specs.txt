//GBV CHUNK vocals


int numVocalPhrases;

struct VocalWord
{
	uint time;//in MS from start
	uint len;//in MS
	short startNote, endNote;
	String value;//the text
}

struct VocalPhrase
{
	uint time;//in MS from start
	byte type;//0=REGULAR,1=BLANK,2=RHYTHM
	int wordsCount;
	switch(type)
	{
		case REGULAR:
			VocalWord[wordsCount] words;

		case BLANK:

		case RHYTHM:
			byte rhythmType;
			uint[wordsCount] rhythmHits;
	}
}

VocalPhrase[numVocalPhrases] phrases;

int[4][5] starlevels;
