struct Vertex
{
	float X, Y, Z;
	float U, V;
};

struct Polygon
{
	float normalX, normalY, normalZ;
	float tangentX, tangentY, tangentZ;
	uint textureIndex;
	uint numVertices;
	Vertex[numVertices] vertices;
};

uint numPolygons;

Polygon[numPolygons] polys;

char '}';

char 'E'; //entities

char '{';

struct Camera
{
	uint cameraIndex;
	uint cameraPart;
	uint cameraType;//matches to Venue.CamBlendPos.TYPE_LEN
	float x, y, z;
	float dirx, diry, dirz;
}

struct SLightTarget
{
	byte type;//not used yet, reserved
	char fade;//a-z, for moving light interpolation (sweeps, etc)
	Vector3 direction;//direction for the light to point
}

struct SLight
{
	uint lightindex;//unique light index
	uint type;//corresponds to Venue.DLight.LIGHT_TYPE
	float innerAngle;//in revolutions (0-1)
	float outerAngle;
	Vector3 position;
	uint numTargets;
	SLightTarget[numTargets] targets;
}

while( ! '}' )
{
	char[3] type;

	switch(type)
	{
	case "cam": //camera data
		uint cameraCount;
		Camera[cameraCount] cameras;
	case "lit": //spotlight data
		uint lightCount;
		SLight[lightCount] sLights;
	}
}
